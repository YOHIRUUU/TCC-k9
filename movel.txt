import React, { useState, useRef } from 'react';
import {
  StyleSheet,
  Text,
  View,
  TextInput,
  TouchableOpacity,
  Alert,
  Animated,
  Platform,
} from 'react-native';

export default function App() {
  const [nome, setNome] = useState('');
  const [email, setEmail] = useState('');
  const [senha, setSenha] = useState('');

  const posX = useRef(new Animated.Value(0)).current;
  const posY = useRef(new Animated.Value(0)).current;
  const fadeConteudo = useRef(new Animated.Value(1)).current;

  function fugirBotao() {
    const novoX = Math.random() * 100 - 50;
    const novoY = Math.random() * 100 - 50;

    Animated.parallel([
      Animated.timing(posX, {
        toValue: novoX,
        duration: 300,
        useNativeDriver: true,
      }),
      Animated.timing(posY, {
        toValue: novoY,
        duration: 300,
        useNativeDriver: true,
      }),
    ]).start();
  }

  function handleSubmit() {
    if (nome === '' || email === '' || senha === '') {
      Alert.alert('Aviso', 'Preencha todos os campos.');
      return;
    }

    Alert.alert(
      'Sucesso',
      `Olá ${nome}, login foi efetuado!`
    );
  }

  return (
    <View style={styles.background}>

      {/* GRADIENTE NO EXPO WEB */}
      {Platform.OS === 'web' && (
        <View
          style={styles.gradientWeb}
        />
      )}

      {/* FUNDO PARA CELULAR */}
      {Platform.OS !== 'web' && (
        <View
          style={styles.gradientMobile}
        />
      )}

      {/* CONTEÚDO */}
      <Animated.View
        style={[
          styles.overlay,
          {
            opacity: fadeConteudo,
          },
        ]}
      >

        <View style={styles.containerInputs}>

          <Text style={styles.label}>
            Usuário:
          </Text>

          <TextInput
            style={styles.input}
            placeholder="Digite seu Nome"
            placeholderTextColor="#999"
            value={nome}
            onChangeText={setNome}
          />

          <Text style={styles.label}>
            E-mail:
          </Text>

          <TextInput
            style={styles.input}
            placeholder="Digite seu E-mail"
            placeholderTextColor="#999"
            keyboardType="email-address"
            value={email}
            onChangeText={setEmail}
          />

          <Text style={styles.label}>
            Senha:
          </Text>

          <TextInput
            style={styles.input}
            placeholder="Digite sua Senha"
            placeholderTextColor="#999"
            secureTextEntry={true}
            value={senha}
            onChangeText={setSenha}
          />

        </View>

        {/* BOTÃO */}
        <Animated.View
          style={{
            width: '100%',
            transform: [
              { translateX: posX },
              { translateY: posY },
            ],
          }}
        >

          <TouchableOpacity
            style={styles.botao}
            onPress={handleSubmit}
          >
            <Text style={styles.textoBotao}>
              Entrar
            </Text>
          </TouchableOpacity>

        </Animated.View>

      </Animated.View>

    </View>
  );
}

const styles = StyleSheet.create({

  background: {
    flex: 1,
    width: '100%',
    height: '100%',
    backgroundColor: '#07152b',
  },

  gradientWeb: {
    position: 'absolute',
    top: 0,
    left: 0,
    right: 0,
    bottom: 0,

    backgroundImage:
      'linear-gradient(135deg, #07152b 0%, #123c69 50%, #d9822b 100%)',
  },

  gradientMobile: {
    position: 'absolute',
    top: 0,
    left: 0,
    right: 0,
    bottom: 0,
    backgroundColor: '#123c69',
  },

  overlay: {
    position: 'absolute',
    width: '100%',
    height: '100%',

    backgroundColor: 'rgba(7, 21, 43, 0.55)',

    paddingHorizontal: 30,

    justifyContent: 'center',
    alignItems: 'center',
  },

  containerInputs: {
    width: '100%',
    marginBottom: 20,
  },

  label: {
    color: '#d9822b',
    fontSize: 14,
    marginBottom: 5,
    alignSelf: 'flex-start',
  },

  input: {
    backgroundColor: '#ffffff',

    width: '100%',

    paddingVertical: 10,
    paddingHorizontal: 15,

    borderRadius: 5,

    marginBottom: 20,

    color: '#333333',
    fontSize: 15,
  },

  botao: {
    backgroundColor: 'transparent',

    borderWidth: 1,
    borderColor: '#d9822b',

    padding: 12,

    borderRadius: 5,

    alignItems: 'center',
    justifyContent: 'center',

    width: '100%',
  },

  textoBotao: {
    color: '#d9822b',

    fontWeight: 'bold',

    fontSize: 16,
  },

});
